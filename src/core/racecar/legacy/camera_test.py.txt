# threshold for red and blue, expressed by array
low_hsv = [[0, 131, 21], [77, 255, 0]]
high_hsv = [[13, 255, 255], [131, 255, 255]]  # RED BLUE


def my_find_contours(src):
    hsv = cv2.cvtColor(src, cv2.COLOR_BGR2HSV)

    lower = np.array(low_hsv)
    higher = np.array(high_hsv)

    dst1 = cv2.inRange(hsv, lower[0], higher[0])
    dst2 = cv2.inRange(hsv, lower[1], higher[1])

    dst = cv2.add(dst1, dst2)

    element3 = np.ones((3, 3), np.uint8)
    # image preprocess #
    # cv2.erode(dst, element3, dst)
    # cv2.erode(dst, element3, dst)
    cv2.erode(dst, element3, dst)
    cv2.dilate(dst, element3, dst)
    cv2.dilate(dst, element3, dst)
    cv2.imshow('dst', dst)
    ret, contours, hierarchy = cv2.findContours(dst, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)

    area_list = []
    center_x = [0]
    for c in contours:
        area = cv2.contourArea(c)
        area_list.append(area)

    if len(area_list) > 0:
        area_threshold = np.mean(area_list)

        for c in contours:
            area = cv2.contourArea(c)
            if area > area_threshold and area > 1000:
                x, y, w, h = cv2.boundingRect(c)
                cv2.rectangle(src, (x, y), (x + w, y + h), (255, 0, 0), 2)
                point = x + w / 2
                cv2.circle(src, (int(point), 240), 3, (0, 0, 0), -1)
                center_x.append(int(point))
                # print("object center is " + str(x + w / 2))

    if len(center_x) > 1:
        center_point = sum(center_x) / (len(center_x) - 1)
    else:
        center_point = 0
    cv2.circle(src, (int(center_point), 240), 5, (0, 255, 0), -1)
    cv2.line(src, (320, 0), (320, 480), (0, 0, 255), 2)
    return center_point
